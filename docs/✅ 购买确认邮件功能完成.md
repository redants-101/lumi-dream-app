```typescript
// ✅ 发送订阅确认邮件
try {
  // 1. 从 Supabase Auth 查询用户信息
  const { data: userData, error: userError } = await supabase.auth.admin.getUserById(userId)
  
  if (userError || !userData.user) {
    // 备选方案：使用 Creem 提供的 customer_email
    if (customer_email) {
      await sendSubscriptionConfirmationEmail({
        to: customer_email,
        userName: customer_email.split('@')[0],
        tier: tierInfo.tier,
        billingCycle: tierInfo.billingCycle,
      })
    }
  } else {
    // 主方案：使用完整的用户信息
    const user = userData.user
    const userEmail = user.email
    const userName = user.user_metadata?.full_name || 
                     user.user_metadata?.name || 
                     userEmail?.split('@')[0] || 
                     "Friend"
    
    if (userEmail) {
      await sendSubscriptionConfirmationEmail({
        to: userEmail,
        userName: userName,
        tier: tierInfo.tier,
        billingCycle: tierInfo.billingCycle,
      })
    }
  }
} catch (emailError) {
  // 邮件发送失败不影响订阅激活
  console.error("⚠️ [Webhook] Error sending confirmation email:", emailError)
}
```

## ✨ 核心特性

### 1. 双重用户信息获取策略

#### 主策略：从 Supabase Auth 查询

```typescript
const { data: userData } = await supabase.auth.admin.getUserById(userId)
```

**优点**：
- ✅ 获取完整用户信息
- ✅ 包含 `user_metadata`（用户名、头像等）
- ✅ 更个性化的邮件内容

**用户名优先级**：
1. `user_metadata.full_name`（GitHub/Google 登录提供）
2. `user_metadata.name`（备选）
3. 邮箱前缀 `email.split('@')[0]`
4. `"Friend"`（最终兜底）

---

#### 备选策略：使用 Creem 提供的邮箱

```typescript
if (customer_email) {
  await sendSubscriptionConfirmationEmail({
    to: customer_email,
    userName: customer_email.split('@')[0],
    // ...
  })
}
```

**触发条件**：
- Supabase 查询失败
- 用户信息不完整
- 权限问题

**优点**：
- ✅ 保证邮件一定能发送
- ✅ 不依赖 Supabase 状态
- ✅ 提高成功率

---

### 2. 完善的错误处理

#### 错误不影响主流程

```typescript
try {
  // 邮件发送逻辑
} catch (emailError) {
  console.error("⚠️ [Webhook] Error sending confirmation email:", emailError)
  // 不影响订阅激活，只记录错误
}
```

**设计理念**：
- ✅ 订阅激活是核心功能，必须成功
- ✅ 邮件发送是附加功能，失败可接受
- ✅ Webhook 仍返回 `200` 状态码
- ✅ 避免 Creem 重复发送 webhook

---

或

**当前**：发送失败后不重试

```typescript
async function sendEmailWithRetry(params, maxRetries = 3) {
  for (let i = 0; i < maxRetries; i++) {
    const success = await sendEmail(params)
    if (success) return true
    
    // 等待 1 秒后重试
    await new Promise(resolve => setTimeout(resolve, 1000 * (i + 1)))
  }
  return false
}
```
